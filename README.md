```python
# 🤖 LLM 은 '해석'만, 숫자는 '데이터'로 — 할루시네이션 없는 RAG
async def answer(question: str, user_id: str):
    plan = sanitize(await llm_planner.plan(question))     # 질문 → 구조화된 조회 조건 (검증 후 사용)
    query = plan.to_query(user_id=user_id)                # 날짜 계산·사용자 격리는 코드가 강제
    result = store.aggregate(query)                       # 합계·건수·평균은 OpenSearch 집계
    async for token in llm_answerer.stream(question, result):
        yield token                                       # LLM 은 계산된 값으로 문장만 생성
```
